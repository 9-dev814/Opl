local HasDrawing = (typeof(Drawing) == "table" and typeof(Drawing.new) == "function")

local function NewLine()
    if HasDrawing then
        local ok, line = pcall(function()
            local l = Drawing.new("Line")
            l.Thickness = 1.5
            l.Color = Color3.fromRGB(235, 65, 65)
            l.Transparency = 0.85
            l.Visible = false
            return l
        end)
        if ok and line then return line, true end
    end
    local f = Instance.new("Frame")
    f.AnchorPoint = Vector2.new(0, 0.5)
    f.BackgroundColor3 = Color3.fromRGB(235, 65, 65)
    f.BorderSizePixel = 0
    f.Size = UDim2.fromOffset(0, 1.5)
    f.Visible = false
    f.Parent = ScreenGui
    return f, false
end

local function SetLine(obj, isDraw, p1, p2, vis)
    if not vis then obj.Visible = false return end
    if isDraw then
        obj.From = p1; obj.To = p2; obj.Visible = true
    else
        local delta = p2 - p1
        local dist = delta.Magnitude
        if dist <= 1 then obj.Visible = false return end
        obj.Position = UDim2.fromOffset(p1.X, p1.Y)
        obj.Size = UDim2.fromOffset(dist, 1.5)
        obj.Rotation = math.deg(math.atan2(delta.Y, delta.X))
        obj.Visible = true
    end
end

--// Core Systems Engine
local function GetAimTarget()
    local vp = Camera.ViewportSize
    local center = Vector2.new(vp.X * 0.5, vp.Y * 0.5)
    local bestPart = nil
    local bestDist = State.FOVRadius

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character then
            local sameTeam = State.TeamCheck and plr.Team and LocalPlayer.Team and (plr.Team == LocalPlayer.Team)
            if not sameTeam then
                local hum = plr.Character:FindFirstChildOfClass("Humanoid")
                local targetPart = plr.Character:FindFirstChild(State.AimPart) or plr.Character:FindFirstChild("Head")
                if hum and hum.Health > 0 and targetPart then
                    local sp, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
                    if onScreen and sp.Z > 0 then
                        local d = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                        if d <= bestDist then
                            bestDist = d
                            bestPart = targetPart
                        end
                    end
                end
            end
        end
    end
    return bestPart
end

-- Main Render Loops
RunService:BindToRenderStep("LunyxAimStep", Enum.RenderPriority.Camera.Value + 1, function()
    local vp = Camera.ViewportSize
    SyncFOV(Vector2.new(vp.X * 0.5, vp.Y * 0.5))

    if State.AimbotEnabled then
        ActiveTarget = GetAimTarget()
        if ActiveTarget and (State.AimKeyHeld or State.AlwaysLock) then
            local goal = CFrame.lookAt(Camera.CFrame.Position, ActiveTarget.Position)
            local alpha = math.clamp(State.AimStrength, 0.1, 1)
            Camera.CFrame = (alpha >= 0.99) and goal or Camera.CFrame:Lerp(goal, alpha)
        end
    end
end)

RunService:BindToRenderStep("LunyxMoveStep", Enum.RenderPriority.Last.Value, function(dt)
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local root = char and char:FindFirstChild("HumanoidRootPart")

    if hum then
        if State.SpeedEnabled then hum.WalkSpeed = State.WalkSpeed end
        if State.JumpPowerEnabled then hum.JumpPower = State.JumpPower end
        if State.SpinEnabled then hum.AutoRotate = false end
    end

    if root and State.SpinEnabled then
        root.CFrame = root.CFrame * CFrame.Angles(0, State.SpinSpeed * dt, 0)
    end

    if State.Noclip and char then
        for _, p in ipairs(char:GetChildren()) do
            if p:IsA("BasePart") then p.CanCollide = false end
        end
    end

    if State.Fullbright then
        Lighting.Brightness = 2
        Lighting.ClockTime = 14
        Lighting.GlobalShadows = false
        Lighting.Ambient = Color3.fromRGB(255, 255, 255)
    end
end)

RunService:BindToRenderStep("LunyxVisualsStep", Enum.RenderPriority.Last.Value, function()
    local vp = Camera.ViewportSize
    local origin = Vector2.new(vp.X * 0.5, vp.Y)

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer and plr.Character then
            local char = plr.Character
            local root = char:FindFirstChild("HumanoidRootPart")
            local hum = char:FindFirstChildOfClass("Humanoid")
            local e = ESPCache[plr]

            if not e then
                local hl = Instance.new("Highlight")
                hl.FillColor = Color3.fromRGB(235, 50, 50)
                hl.OutlineColor = Color3.fromRGB(255, 255, 255)
                hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                hl.Parent = char

                local bb = Instance.new("BillboardGui")
                bb.Size = UDim2.new(0, 120, 0, 22)
                bb.StudsOffset = Vector3.new(0, 2.3, 0)
                bb.AlwaysOnTop = true
                bb.Adornee = root
                bb.Parent = root

                local lbl = Instance.new("TextLabel")
                lbl.Size = UDim2.new(1, 0, 1, 0)
                lbl.BackgroundTransparency = 1
                lbl.TextColor3 = Color3.fromRGB(255, 95, 95)
                lbl.Font = Enum.Font.GothamBold
                lbl.TextSize = 11
                lbl.Parent = bb

                local tracer, isDraw = NewLine()
                e = { hl = hl, bb = bb, lbl = lbl, tracer = tracer, isDraw = isDraw }
                ESPCache[plr] = e
            end

            e.hl.Enabled = State.ChamsEnabled
            e.bb.Enabled = State.ESPEnabled

            if State.ESPEnabled and root and hum then
                local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                local dist = myRoot and math.floor((myRoot.Position - root.Position).Magnitude) or 0
                e.lbl.Text = string.format("%s [%d HP | %dm]", plr.DisplayName, math.floor(hum.Health), dist)
            end

            if State.TracersEnabled and root then
                local sp, onScreen = Camera:WorldToViewportPoint(root.Position)
                if onScreen and sp.Z > 0 then
                    SetLine(e.tracer, e.isDraw, origin, Vector2.new(sp.X, sp.Y), true)
                else
                    SetLine(e.tracer, e.isDraw, Vector2.zero, Vector2.zero, false)
                end
            else
                SetLine(e.tracer, e.isDraw, Vector2.zero, Vector2.zero, false)
            end
        end
    end
end)
